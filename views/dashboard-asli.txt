<script>
    // --- TOGGLE PRIVASI WALLET ---
    function togglePrivasi() {
        const val = document.getElementById('valSaldo');
        const label = document.getElementById('labelSaldo');
        const eye = document.getElementById('toggleEye');
        const profit = document.getElementById('profitSection');
        const profitHidden = document.getElementById('profitHidden');

        if (val.style.display === 'none') {
            val.style.display = 'inline';
            label.style.display = 'none';
            eye.classList.replace('bi-eye-slash', 'bi-eye');
            if(profit) profit.style.display = 'flex';
            if(profitHidden) profitHidden.style.display = 'none';
        } else {
            val.style.display = 'none';
            label.style.display = 'inline';
            eye.classList.replace('bi-eye', 'bi-eye-slash');
            if(profit) profit.style.display = 'none';
            if(profitHidden) profitHidden.style.display = 'block';
        }
    }

   // --- ALERT UPGRADE ---

            showCancelButton: true, confirmButtonColor: '#28a745', cancelButtonColor: '#6c757d',
            confirmButtonText: '<i class="bi bi-rocket-takeoff"></i> Hubungi Developer',
            cancelButtonText: 'Tutup', reverseButtons: true

    // --- INISIALISASI CHART.JS (REAL DATA) ---
    document.addEventListener("DOMContentLoaded", function() {
        
        // 1. TANGKAP DATA DARI BRANKAS TEXTAREA
        const labelBulan = JSON.parse(document.getElementById('val-bulan').value);
        const dataPemasukan = JSON.parse(document.getElementById('val-masuk').value);
        const dataPengeluaran = JSON.parse(document.getElementById('val-keluar').value);
        const dataProfit = JSON.parse(document.getElementById('val-profit').value);

        // Formatter Rupiah untuk Tooltip & Sumbu Y
        const rupiahFormatter = function(value) {
            if (value >= 1000000) return 'Rp ' + (value / 1000000).toFixed(1) + 'Jt';
            if (value >= 1000) return 'Rp ' + (value / 1000).toFixed(0) + 'Rb';
            return 'Rp ' + value;
        };

        // 2. GRAFIK ARUS KAS (Model Line/Area Chart Mulus)
        const ctxFinance = document.getElementById('financeChart').getContext('2d');
        new Chart(ctxFinance, {
            type: 'line', 
            data: {
                labels: labelBulan,
                datasets: [
                    {
                        label: 'Pemasukan',
                        data: dataPemasukan,
                        borderColor: '#2ecc71', // Hijau
                        backgroundColor: 'rgba(46, 204, 113, 0.15)', // Fill hijau transparan
                        borderWidth: 3,
                        pointBackgroundColor: '#2ecc71',
                        pointBorderWidth: 2,
                        pointRadius: 4,
                        fill: true, // Mengaktifkan efek warna area di bawah garis
                        tension: 0.4 // Membuat garis melengkung halus
                    },
                    {
                        label: 'Pengeluaran',
                        data: dataPengeluaran,
                        borderColor: '#ff4757', // Merah
                        backgroundColor: 'rgba(255, 71, 87, 0.1)', // Fill merah transparan
                        borderWidth: 3,
                        pointBackgroundColor: '#ff4757',
                        pointBorderWidth: 2,
                        pointRadius: 4,
                        fill: true,
                        tension: 0.4
                    }
                ]
            },
            options: {
                responsive: true, maintainAspectRatio: false,
                interaction: { mode: 'index', intersect: false }, // Tooltip muncul bersamaan
                scales: {
                    y: { beginAtZero: true, ticks: { callback: rupiahFormatter, font: {size: 10} } },
                    x: { grid: { display: false } }
                },
                plugins: { legend: { position: 'top', labels: { boxWidth: 12, usePointStyle: true, font: {size: 11} } } }
            }
        });

        // 3. GRAFIK PROFIT BULANAN (Model Bar Chart)
        const ctxProfit = document.getElementById('profitChart').getContext('2d');
        new Chart(ctxProfit, {
            type: 'bar',
            data: {
                labels: labelBulan,
                datasets: [{
                    label: 'Profit Bersih',
                    data: dataProfit,
                    backgroundColor: function(context) {
                        const value = context.dataset.data[context.dataIndex];
                        return value < 0 ? '#ff4757' : '#3498db'; // Merah jika rugi, Biru jika untung
                    },
                    borderRadius: 6, // Ujung batang melengkung elegan
                    borderWidth: 0
                }]
            },
            options: {
                responsive: true, maintainAspectRatio: false,
                scales: {
                    y: { beginAtZero: true, ticks: { callback: rupiahFormatter, font: {size: 10} } },
                    x: { grid: { display: false } }
                },
                plugins: { 
                    legend: { display: false }, // Disembunyikan karena hanya ada 1 dataset
                    tooltip: {
                        callbacks: {
                            label: function(context) {
                                return 'Profit: Rp ' + context.raw.toLocaleString('id-ID');
                            }
                        }
                    }
                }
            }
        });
    });
</script>
